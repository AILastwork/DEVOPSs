git init — инициализировать (создать) локальный репозиторий в папке с проектом

git config --global user.email "почта_аккаунта" — указать email автора коммитов
git config --global user.name "Имя" — указать имя автора коммитов

git branch -m main - переименовать мастер ветку в  main
git status — посмотреть текущее состояние файлов (измененные, новые, подготовленные)
git add . — подготовить все изменения к коммиту (или указать конкретный файл: git add имя_файла)
git commit -m "описание_коммита" — зафиксировать изменения (сделать точку сохранения)

git remote add origin url.git — привязать удаленный репозиторий под именем origin
git remote -v — посмотреть привязанные адреса (URL) удаленных репозиториев
git push -u origin main — отправить коммиты в удаленный репозиторий (ветка main) и запомнить связь

git clone url.git — скачать (склонировать) удаленный репозиторий к себе на ПК
.gitignore - файл в котором указываем какие файлы не отслеживаются  гитом
